// --- PRAWDZIWE ROZPOZNAWANIE AI --- async function identifyPlantAI(base64Image) { const apiKey = "TWÓJ_KLUCZ_API"; // <--- TUTAJ WKLEJ SWÓJ KLUCZ Z PLANT.ID if (apiKey === "TWÓJ_KLUCZ_API") { alert("BŁĄD: Nie wpisałeś klucza API w kodzie!"); return; } const data = { images: [base64Image], latitude: 49.207, longitude: 16.608, similar_images: true, }; try { const response = await fetch("https://plant.id/api/v3/identification", { method: "POST", headers: { "Api-Key": apiKey, "Content-Type": "application/json", }, body: JSON.stringify(data), }); const result = await response.json(); processAIResult(result); } catch (error) { console.error("Błąd AI:", error); alert("AI nie mogło przetworzyć zdjęcia. Sprawdź połączenie."); } } function processAIResult(result) { // Sprawdzamy, czy AI coś znalazło if (result.suggestions && result.suggestions.length > 0) { const bestMatch = result.suggestions[0]; const probability = Math.round(bestMatch.probability * 100); if (probability < 10) { alert("AI mówi: To nie wygląda na roślinę! (Pewność: " + probability + "%)"); return; } const plantName = bestMatch.plant_name; // Wyświetlamy wynik if (confirm("AI rozpoznało: " + plantName + "\nPewność: " + probability + "%\nCzy chcesz wyszukać tę roślinę w bibliotece?")) { changeSection('home'); document.getElementById('searchInput').value = plantName; renderPlants(plantName); } } else { alert("AI nie rozpoznało żadnej rośliny na tym zdjęciu."); } } // Funkcja przechwytująca klatkę z wideo i zamieniająca na Base64 function captureAndIdentify() { const video = document.getElementById('video'); const canvas = document.createElement('canvas'); canvas.width = video.videoWidth; canvas.height = video.videoHeight; canvas.getContext('2d').drawImage(video, 0, 0); const base64Image = canvas.toDataURL('image/jpeg'); document.getElementById('scanLine').classList.remove('hidden'); identifyPlantAI(base64Image); }
